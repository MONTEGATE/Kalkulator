function calculate() { // 1. Pokupi vrednosti iz forme const fenceType = document.getElementById('fenceType').value; const length = parseFloat(document.getElementById('length').value) || 0; const height = document.getElementById('height').value; const city = document.getElementById('city').value; const hasDelivery = document.getElementById('delivery').checked; const hasInstallation = document.getElementById('installation').checked; const gate = document.getElementById('gate').value; // 2. Ažuriraj sliku na osnovu izabranog tipa ograde const fencePreview = document.getElementById('fencePreview'); const fenceImages = { '3D Panel': 'images/3d-panel.jpg', 'Dekorativna': 'images/dekorativna.jpg', 'Z-Roletna': 'images/z-roletna.jpg', 'CNC Laserski rezana': 'images/cnc.jpg' }; if (fenceImages[fenceType]) { fencePreview.src = fenceImages[fenceType]; } // 3. Prikaži podatke u Live Summary delu na dnu document.getElementById('summaryType').innerText = fenceType; document.getElementById('summaryLength').innerText = length + ' m'; document.getElementById('summaryHeight').innerText = height; document.getElementById('summaryCity').innerText = city; document.getElementById('summaryExtras').innerText = (hasDelivery ? 'Dostava ' : '') + (hasInstallation ? 'Ugradnja' : (!hasDelivery && !hasInstallation ? 'Samo materijal' : '')); // 4. Pripremi tekst za Viber, WhatsApp i SMS dugmiće const phone = document.getElementById('phoneNumber').value.trim(); const message = `Pozdrav! Želim ponudu sa sajta MONTEGATE:\n` + `- Tip ograde: ${fenceType}\n` + `- Dužina: ${length}m\n` + `- Visina: ${height}\n` + `- Kapija: ${gate}\n` + `- Dostava: ${hasDelivery ? 'Da' : 'Ne'}\n` + `- Ugradnja: ${hasInstallation ? 'Da' : 'Ne'}\n` + `- Grad: ${city}`; const encodedMessage = encodeURIComponent(message); // Ažuriraj linkove dugmića document.getElementById('whatsappBtn').href = `https://wa.me/${phone}?text=${encodedMessage}`; document.getElementById('viberBtn').href = `viber://chat?number=%2B${phone}&text=${encodedMessage}`; document.getElementById('smsBtn').href = `sms:${phone}?body=${encodedMessage}`; } // Pokreni funkciju automatski čim se stranica učita window.onload = calculate;
